include(FetchContent)

FetchContent_Declare(
    Catch2
    GIT_REPOSITORY https://github.com/catchorg/Catch2.git
    GIT_TAG        v3.7.1
)
FetchContent_MakeAvailable(Catch2)

add_executable(veksel_tests
    test_log.cpp
    test_compiler.cpp
    test_swap.cpp
    test_soak.cpp
)

target_link_libraries(veksel_tests PRIVATE
    veksel_core
    Catch2::Catch2WithMain
)

include(Catch)
catch_discover_tests(veksel_tests)
